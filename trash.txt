def get_marks_controller():
    student_info = []

    # Get all marks records
    data = Marks.query.all()
    marks = []

    for i in data:
        marks.append({
            "student_id": i.student_id,
            "subject_id": i.subject_id,
            "student_marks": i.marks
        })

    # Get all students
    students = Student.query.all()
    student_list = []

    for i in students:
        student_list.append({
            "student_id": i.id,
            "student_name": i.name,
            "student_rollno": i.rollno,
            "student_class": i.class_name
        })

    # Match marks with students
    for mark in marks:
        for student in student_list:

            if mark["student_id"] == student["student_id"]:

                # Check whether the student already exists
                existing_student = None

                for s in student_info:
                    if s["student_id"] == student["student_id"]:
                        existing_student = s
                        break

                # If the student doesn't exist, add them
                if existing_student is None:
                    existing_student = {
                        "student_id": student["student_id"],
                        "student_name": student["student_name"],
                        "student_rollno": student["student_rollno"],
                        "student_class": student["student_class"],
                        "subjects": []
                    }

                    student_info.append(existing_student)

                # Add this subject's marks to the student
                existing_student["subjects"].append({
                    "subject_id": mark["subject_id"],
                    "student_marks": mark["student_marks"]
                })

    return jsonify(student_info)
